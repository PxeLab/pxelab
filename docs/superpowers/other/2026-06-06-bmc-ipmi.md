```go
import (

func (d *DB) CreateBMCConfig(cfg *models.BMCConfig) error {
    res, err := d.DB.Exec(
        "INSERT INTO bmc_configs (device_id, host, port, username, password, redfish) VALUES (?, ?, ?, ?, ?, ?)",
        cfg.DeviceID, cfg.Host, cfg.Port, cfg.Username, cfg.Password, cfg.Redfish,
    )
    if err != nil { return err }
    cfg.ID, _ = res.LastInsertId()
    return nil
}

func (d *DB) GetBMCConfig(id int64) (*models.BMCConfig, error) {
    row := d.DB.QueryRow("SELECT id, device_id, host, port, username, password, redfish, created_at FROM bmc_configs WHERE id = ?", id)
    cfg := &models.BMCConfig{}
    err := row.Scan(&cfg.ID, &cfg.DeviceID, &cfg.Host, &cfg.Port, &cfg.Username, &cfg.Password, &cfg.Redfish, &cfg.CreatedAt)
    if err == sql.ErrNoRows { return nil, fmt.Errorf("bmc config not found") }
    return cfg, err
}

func (d *DB) GetBMCConfigByDevice(deviceID int64) (*models.BMCConfig, error) {
    row := d.DB.QueryRow("SELECT id, device_id, host, port, username, password, redfish, created_at FROM bmc_configs WHERE device_id = ?", deviceID)
    cfg := &models.BMCConfig{}
    err := row.Scan(&cfg.ID, &cfg.DeviceID, &cfg.Host, &cfg.Port, &cfg.Username, &cfg.Password, &cfg.Redfish, &cfg.CreatedAt)
    if err == sql.ErrNoRows { return nil, fmt.Errorf("bmc config not found") }
    return cfg, err
}

func (d *DB) ListBMCConfigs() ([]models.BMCConfig, error) {
    rows, err := d.DB.Query("SELECT id, device_id, host, port, username, password, redfish, created_at FROM bmc_configs ORDER BY id DESC")
    if err != nil { return nil, err }
    defer rows.Close()
    var cfgs []models.BMCConfig
    for rows.Next() {
        var cfg models.BMCConfig
        if err := rows.Scan(&cfg.ID, &cfg.DeviceID, &cfg.Host, &cfg.Port, &cfg.Username, &cfg.Password, &cfg.Redfish, &cfg.CreatedAt); err != nil {
            return nil, err
        }
        cfgs = append(cfgs, cfg)
    }
    return cfgs, rows.Err()
}

func (d *DB) UpdateBMCConfig(cfg *models.BMCConfig) error {
    _, err := d.DB.Exec(
        "UPDATE bmc_configs SET host = ?, port = ?, username = ?, password = ?, redfish = ? WHERE id = ?",
        cfg.Host, cfg.Port, cfg.Username, cfg.Password, cfg.Redfish, cfg.ID,
    )
    return err
}

func (d *DB) DeleteBMCConfig(id int64) error {
    _, err := d.DB.Exec("DELETE FROM bmc_configs WHERE id = ?", id)
    return err
}

// ListBMCConfigsWithDevice 联表查询，返回 BMC 配置 + 设备名
func (d *DB) ListBMCConfigsWithDevice() ([]map[string]interface{}, error) {
    rows, err := d.DB.Query(`
        SELECT c.id, c.device_id, c.host, c.port, c.username, c.redfish, c.created_at,
               d.hostname, d.mac
        FROM bmc_configs c LEFT JOIN devices d ON c.device_id = d.id
        ORDER BY c.id DESC
    `)
    if err != nil { return nil, err }
    defer rows.Close()
    var result []map[string]interface{}
    for rows.Next() {
        var id, deviceID, port int64
        var host, username string
        var redfish bool
        var createdAt time.Time
        var hostname, mac *string
        err := rows.Scan(&id, &deviceID, &host, &port, &username, &redfish, &createdAt, &hostname, &mac)
        if err != nil { return nil, err }
        item := map[string]interface{}{
            "id": id, "device_id": deviceID, "host": host, "port": port,
            "username": username, "redfish": redfish, "created_at": createdAt,
        }
        if hostname != nil { item["device_hostname"] = *hostname }
        if mac != nil { item["device_mac"] = *mac }
        result = append(result, item)
    }
    return result, rows.Err()
}
```

```go
// internal/bmc/types.go
package bmc

import "context"

type PowerState string
const (
    PowerOn     PowerState = "on"
    PowerOff    PowerState = "off"
    PowerStatusUnknown PowerState = "unknown"
)

type BootDevice string
const (
    BootPXE   BootDevice = "pxe"
    BootDisk  BootDevice = "disk"
    BootCDROM BootDevice = "cdrom"
    BootBIOS  BootDevice = "bios"
)

type Config struct {
    Host     string
    Port     int
    Username string
    Password string
    Redfish  bool
}

type Controller interface {
    PowerOn(ctx context.Context) error
    PowerOff(ctx context.Context) error
    PowerRestart(ctx context.Context) error
    PowerStatus(ctx context.Context) (PowerState, error)
    SetBootDevice(ctx context.Context, device BootDevice) error
    Close() error
```

```go
// internal/bmc/ipmi.go
package bmc

import (
    "context"
    "fmt"
    "net"

    "github.com/bougou/go-ipmi"
)

type ipmiController struct {
    client *ipmi.Client
    cfg    Config
}

func NewIPMIController(cfg Config) (Controller, error) {
    addr := net.JoinHostPort(cfg.Host, fmt.Sprintf("%d", cfg.Port))
    client, err := ipmi.NewClient(addr, cfg.Username, cfg.Password)
    if err != nil {
        return nil, fmt.Errorf("create ipmi client: %w", err)
    }
    return &ipmiController{client: client, cfg: cfg}, nil
}

func (c *ipmiController) Close() error {
    if c.client != nil {
        return c.client.Close()
    }
    return nil
}

func (c *ipmiController) PowerOn(ctx context.Context) error {
    return c.client.ChassisControl(ipmi.ChassisControlPowerUp)
}

func (c *ipmiController) PowerOff(ctx context.Context) error {
    return c.client.ChassisControl(ipmi.ChassisControlPowerDown)
}

func (c *ipmiController) PowerRestart(ctx context.Context) error {
    return c.client.ChassisControl(ipmi.ChassisControlPowerCycle)
}

func (c *ipmiController) PowerStatus(ctx context.Context) (PowerState, error) {
    s, err := c.client.ChassisStatus()
    if err != nil {
        return PowerStatusUnknown, fmt.Errorf("get chassis status: %w", err)
    }
    if s.PowerOn {
        return PowerOn, nil
    }
    return PowerOff, nil
}

func (c *ipmiController) SetBootDevice(ctx context.Context, device BootDevice) error {
    var dev ipmi.BootDevice
    switch device {
    case BootPXE:
        dev = ipmi.BootDeviceForcePXE
    case BootDisk:
        dev = ipmi.BootDeviceForceHDD
    case BootCDROM:
        dev = ipmi.BootDeviceForceCDROM
    case BootBIOS:
        dev = ipmi.BootDeviceForceBIOSSetup
    default:
        return fmt.Errorf("unknown boot device: %s", device)
    }
    return c.client.SetBootParam(dev, true)
}
```

```go
// internal/bmc/redfish.go
package bmc

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
    "strings"
)

type redfishController struct {
    cfg     Config
    sysURL  string
    client  *http.Client
}

func NewRedfishController(cfg Config) (Controller, error) {
    c := &redfishController{
        cfg:    cfg,
        client: &http.Client{},
    }
    // Discover System ID
    rootURL := fmt.Sprintf("https://%s:%d", cfg.Host, cfg.Port)
    sysID, err := c.discoverSystem(rootURL)
    if err != nil {
        return nil, fmt.Errorf("discover redfish system: %w", err)
    }
    c.sysURL = fmt.Sprintf("%s/redfish/v1/Systems/%s", rootURL, sysID)
    return c, nil
}

func (c *redfishController) Close() error { return nil }

func (c *redfishController) do(method, url string, body interface{}) (*http.Response, error) {
    var r io.Reader
    if body != nil {
        data, _ := json.Marshal(body)
        r = bytes.NewReader(data)
    }
    req, err := http.NewRequest(method, url, r)
    if err != nil { return nil, err }
    req.SetBasicAuth(c.cfg.Username, c.cfg.Password)
    req.Header.Set("Content-Type", "application/json")
    return c.client.Do(req)
}

func (c *redfishController) discoverSystem(rootURL string) (string, error) {
    resp, err := c.do("GET", rootURL+"/redfish/v1/Systems", nil)
    if err != nil { return "", err }
    defer resp.Body.Close()
    var result struct {
        Members []struct{ OdataID string `json:"@odata.id"` } `json:"Members"`
    }
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return "", err
    }
    if len(result.Members) == 0 { return "", fmt.Errorf("no systems found") }
    parts := strings.Split(result.Members[0].OdataID, "/")
    return parts[len(parts)-1], nil
}

func (c *redfishController) PowerOn(ctx context.Context) error {
    action := map[string]string{"ResetType": "On"}
    resp, err := c.do("POST", c.sysURL+"/Actions/ComputerSystem.Reset", action)
    if err != nil { return err }
    resp.Body.Close()
    return nil
}

func (c *redfishController) PowerOff(ctx context.Context) error {
    action := map[string]string{"ResetType": "GracefulShutdown"}
    resp, err := c.do("POST", c.sysURL+"/Actions/ComputerSystem.Reset", action)
    if err != nil { return err }
    resp.Body.Close()
    return nil
}

func (c *redfishController) PowerRestart(ctx context.Context) error {
    action := map[string]string{"ResetType": "ForceRestart"}
    resp, err := c.do("POST", c.sysURL+"/Actions/ComputerSystem.Reset", action)
    if err != nil { return err }
    resp.Body.Close()
    return nil
}

func (c *redfishController) PowerStatus(ctx context.Context) (PowerState, error) {
    resp, err := c.do("GET", c.sysURL, nil)
    if err != nil { return PowerStatusUnknown, err }
    defer resp.Body.Close()
    var result struct {
        PowerState string `json:"PowerState"`
    }
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return PowerStatusUnknown, err
    }
    if result.PowerState == "On" { return PowerOn, nil }
    return PowerOff, nil
}

func (c *redfishController) SetBootDevice(ctx context.Context, device BootDevice) error {
    var target string
    switch device {
    case BootPXE:
        target = "Pxe"
    case BootDisk:
        target = "HardDisk"
    case BootCDROM:
        target = "Cd"
    case BootBIOS:
        target = "BiosSetup"
    default:
        return fmt.Errorf("unknown boot device: %s", device)
    }
    patch := map[string]interface{}{
        "Boot": map[string]string{
            "BootSourceOverrideTarget": target,
            "BootSourceOverrideEnabled": "Once",
        },
    }
    resp, err := c.do("PATCH", c.sysURL, patch)
    if err != nil { return err }
    resp.Body.Close()
    return nil
}
```

```go
func (s *Server) getBMCConfig(c echo.Context) error {
    id, err := parseID(c)
    if err != nil {
        return c.JSON(http.StatusBadRequest, map[string]string{"error": "invalid id"})
    }
    cfg, err := s.db.GetBMCConfig(id)
    if err != nil {
        return c.JSON(http.StatusNotFound, map[string]string{"error": err.Error()})
    }
    return c.JSON(http.StatusOK, cfg)
}

func (s *Server) createBMCConfig(c echo.Context) error {
    var cfg models.BMCConfig
    if err := c.Bind(&cfg); err != nil {
        return c.JSON(http.StatusBadRequest, map[string]string{"error": err.Error()})
    }
    if cfg.Port == 0 {
        cfg.Port = 623
    }
    if err := s.db.CreateBMCConfig(&cfg); err != nil {
        return c.JSON(http.StatusInternalServerError, map[string]string{"error": err.Error()})
    }
    return c.JSON(http.StatusCreated, &cfg)
}

func (s *Server) updateBMCConfig(c echo.Context) error {
    id, err := parseID(c)
    if err != nil {
        return c.JSON(http.StatusBadRequest, map[string]string{"error": "invalid id"})
    }
    cfg, err := s.db.GetBMCConfig(id)
    if err != nil {
        return c.JSON(http.StatusNotFound, map[string]string{"error": err.Error()})
    }
    if err := c.Bind(cfg); err != nil {
        return c.JSON(http.StatusBadRequest, map[string]string{"error": err.Error()})
    }
    cfg.ID = id
    if err := s.db.UpdateBMCConfig(cfg); err != nil {
        return c.JSON(http.StatusInternalServerError, map[string]string{"error": err.Error()})
    }
    return c.JSON(http.StatusOK, cfg)
}

func (s *Server) deleteBMCConfig(c echo.Context) error {
    id, err := parseID(c)
    if err != nil {
        return c.JSON(http.StatusBadRequest, map[string]string{"error": "invalid id"})

    if err := s.db.DeleteBMCConfig(id); err != nil {
        return c.JSON(http.StatusInternalServerError, map[string]string{"error": err.Error()})
```

```vue
async function refreshStatus(cfg) {
  try {
    const { data } = await api.get(`/bmc/${cfg.id}/status`)
    statusMap.value[cfg.id] = data.status
  } catch {
    statusMap.value[cfg.id] = 'unknown'
  }
}

function statusClass(cfg) {
  const s = statusMap.value[cfg.id]
  if (s === 'on') return 'active'
  if (s === 'off') return ''
  return 'pending'
}

function statusText(cfg) {
  const s = statusMap.value[cfg.id]
  if (s === 'on') return '开机'
  if (s === 'off') return '关机'
  return '未知'
}

function toggleAll() {
  selectedIds.value = selectAll.value ? configs.value.map(c => c.id) : []
}

async function singleAction(configId, action) {
  try {
    await api.post(`/bmc/${configId}/${action}`)
    // Refresh status after action
    const cfg = configs.value.find(c => c.id === configId)
    if (cfg) setTimeout(() => refreshStatus(cfg), 2000)
  } catch (e) {
    alert('操作失败: ' + (e.response?.data?.error || e.message))
  }
}

async function batchOperation(action) {
  if (selectedIds.value.length === 0) {
    alert('请先选择设备')
    return
  }
  try {
    const { data } = await api.post(`/bmc/batch/${action}`, { config_ids: selectedIds.value })
    // Refresh all statuses after batch
    setTimeout(loadConfigs, 2000)
  } catch (e) {
    alert('批量操作失败: ' + (e.response?.data?.error || e.message))
  }
}

async function importCSV() {
  const lines = csvText.value.trim().split('\n')
  let success = 0, fail = 0
  for (const line of lines) {
    const parts = line.split(',').map(s => s.trim())
    if (parts.length < 5) { fail++; continue }
    try {
      await api.post('/bmc/configs', {
        device_id: 0,
        host: parts[1],
        port: parseInt(parts[2]) || 623,
        username: parts[3],
        password: parts[4],
        redfish: parts[5] === 'true',
        // Match device by MAC if provided
        ...(parts[0] && parts[0] !== 'mac' ? { device_id: 0 } : {})
      })
      success++
    } catch { fail++ }
  }
  alert(`导入完成：成功 ${success}，失败 ${fail}`)
  showImportModal.value = false
  loadConfigs()
}

    bootDeviceCfg.value = null
  } catch (e) {
    alert('设置失败: ' + (e.response?.data?.error || e.message))
```
